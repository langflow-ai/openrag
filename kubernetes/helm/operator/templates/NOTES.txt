To verify the installation:

  kubectl get deployment -n {{ .Release.Namespace }} {{ include "bomarag-operator.fullname" . }}
  kubectl get pods -n {{ .Release.Namespace }} -l control-plane=controller-manager

To check the operator logs:

  kubectl logs -n {{ .Release.Namespace }} -l control-plane=controller-manager -f

The operator will watch for BomaRAG custom resources. To create an BomaRAG instance:

  kubectl apply -f - <<EOF
  apiVersion: bomalogic.com/v1alpha1
  kind: BomaRAG
  metadata:
    name: my-bomarag
    namespace: default
  spec:
    frontend:
      image: bomalogic/bomarag-frontend:latest
    backend:
      image: bomalogic/bomarag-backend:latest
      envSecret: my-backend-env
      storage:
        enabled: true
        size: 10Gi
    langflow:
      image: bomalogic/bomarag-langflow:latest
      envSecret: my-langflow-env
      storage:
        enabled: true
        size: 10Gi
    opensearch:
      host: opensearch.default.svc.cluster.local
      credentialsSecret: opensearch-credentials
  EOF
